    users {
        INTEGER id PK
        TEXT email UK
        TEXT passwordHash
        TEXT mfaSecret
        INTEGER mfaEnabled
        TEXT step3Token
    }
    sessions {
        TEXT id PK
        INTEGER userId FK
        TEXT tokenHash UK
        INTEGER expiresAt
        INTEGER createdAt
    }
    conversion_jobs {
        INTEGER id PK
        INTEGER userId FK
        TEXT originalFilename
        TEXT sourceType
        TEXT targetType
        TEXT status
        INTEGER createdAt
    }
    audit_logs {
        INTEGER id PK
        INTEGER userId FK
        TEXT action
        TEXT ipAddress
        INTEGER createdAt
    }
